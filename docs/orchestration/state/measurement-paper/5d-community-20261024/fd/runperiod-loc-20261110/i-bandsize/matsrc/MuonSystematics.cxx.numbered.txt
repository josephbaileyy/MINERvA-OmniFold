    67	  template<typename T>
    68	  double PlotUtils::MuonUniverseMinerva<T>::GetMuonMomentumShiftMinerva() const {
    69	  
    70	    double muon_shift_minerva;
    71	  
    72	    //--------------------------------------------------
    73	    // Uncertainty on Minerva component of muon momentum
    74	    //--------------------------------------------------
    75	  
    76	    double vertex_Z = T::GetVertexZ();
    77	    double dEdxUncertainty = 0.0;
    78	    double materialAssayUncertainty = 0.0;
    79	    // Use a different correction in the nuclear targets
    80	    if(vertex_Z<NSFDefaults::nuclearTargetZEnd){
    81	      dEdxUncertainty = NSFDefaults::dEdxUncertainty;
    82	      materialAssayUncertainty = NSFDefaults::MaterialAssayUncertainty;
    83	    }
    84	    else{
    85	      dEdxUncertainty = NSFDefaults::dEdxUncertaintyNoNuke;
    86	      materialAssayUncertainty = NSFDefaults::MaterialAssayUncertaintyNoNuke;
    87	    } 
    88	    muon_shift_minerva = sqrt(
    89	      dEdxUncertainty*dEdxUncertainty +
    90	      materialAssayUncertainty*materialAssayUncertainty );
    91	   
    92	    double shift_val_minerva = T::m_nsigma * muon_shift_minerva;
    93	    
    94	    return shift_val_minerva;
    95	  }

    99	  
   100	    double shift_val = GetMuonMomentumShiftMinerva();
   101	  
   102	    return shift_val+T::GetPmuMinerva();

   120	  template<typename T>
   121	  double PlotUtils::MuonUniverseMinos<T>::GetMuonMomentumShiftMinos() const {
   122	  
   123	    double muon_shift_minos;
   124	  
   125	    //--------------------------------------------------
   126	    // Uncertainty on Minos components of muon momentum
   127	    //--------------------------------------------------
   128	  
   129	    // Was Minos component of muon momentum reconstructed by range or curvature
   130	    std::string minosRangeCurveBool = T::GetAnaToolName()+"_minos_used_curvature";
   131	    bool minosRecoByCurve = T::GetBool(minosRangeCurveBool.c_str());
   132	  
   133	    // Sys shifts are made relative to the total (minos component of the) muon momentum
   134	    double minosCVMomentum = T::GetPmuMinos();
   135	    // Uncertainty due to range on Minos component of muon momentum
   136	    double minosRangeUncertainty = minosCVMomentum*NSFDefaults::MinosMuonPRange_Err;
   137	    // Uncertainty due to curvature on Minos component of muon momentum
   138	    double minosCurveUncertainty = 0.0;
   139	    double localCurveUncertainty = 0.0;
   140	    if(minosRecoByCurve){
   141	      localCurveUncertainty = minosCVMomentum>1000.0?NSFDefaults::MinosMuonHighPCurvature_Err:NSFDefaults::MinosMuonLowPCurvature_Err;
   142	      minosCurveUncertainty = minosCVMomentum*localCurveUncertainty;
   143	    }
   144	  
   145	    muon_shift_minos = sqrt(
   146	        minosRangeUncertainty*minosRangeUncertainty +
   147	        minosCurveUncertainty*minosCurveUncertainty );
   148	  
   149	    double shift_val_minos = T::m_nsigma * muon_shift_minos;
   150	    
   151	    return shift_val_minos;
   152	  }

   156	  
   157	    double shift_val = GetMuonMomentumShiftMinos();
   158	  
   159	    return shift_val+T::GetPmuMinos();
   160	  }

   162	  template <class T>
   163	  double PlotUtils::MuonUniverseMinos<T>::GetFluxAndCVWeight(double Enu,
   164	                                                             int nu_pdg) const {
   165	    //Pretty unique in the MAT to have a lateral shift and a vertical shift.  
   166	    //This universe represents a shift in a reconstructed quantity, however
   167	    //we try to cancel out the uncertainties in the flux for the efficiency correction,
   168	    //So does the efficiency denominator also need this shift in the flux weight?
   169	    //I'm putting this in little piece of code, but not enabling it. L. Fields feels it's safest to shift this everywhere
   170	    //if( T::IsTruth() ) return 1;  
   171	
   172	    // If this playlist is not ME, `GetFluxAndCVWeight` shouldn't change
   173	    if (!T::IsPlaylistME(T::GetPlaylist())) return T::GetFluxAndCVWeight();
   174	    // If this playlist is ME RHC, `GetFluxAndCVWeight` shouldn't change
   175	    // This will change if/when RHC correlated fluxes become available
   176	    //if (!T::isFHC()) return T::GetFluxAndCVWeight(); Commented out now that RHC correlated fluxes are available as in Amit's CVS commits 1.19/20 -David L.
   177	
   178	    if (Enu == -99.)   Enu    = T::GetDouble("mc_incomingE")*1e-3;
   179	    if (nu_pdg == -99) nu_pdg = T::GetInt("mc_incoming");
   180	    int universe = T::m_nsigma < 0 ? 0 : 1;
   181	    // This universes's flux weight is modified because the flux is strongly correlated with the muon energy scale.
   182	    // The fluxes used for these systematic universes were extracted as part of Amit's wiggle studies.
   183	    double wgtMod = PlotUtils::flux_reweighter(
   184	      T::GetPlaylist(),nu_pdg,T::UseNuEConstraint(),T::GetNFluxUniverses()).GetSysUniFluxWeightCorrection(Enu,nu_pdg,"Muon_Energy_MINOS",universe); // Putting in Amit's CVS commit 1.19/20 changes -David L.
   185	    return wgtMod*T::GetFluxAndCVWeight();
   186	  }

   188	  template <class T>
   189	  double PlotUtils::MuonUniverseMinos<T>::GetWeightRatioToCV() const {
   190	    // If this playlist is not ME, `GetFluxAndCVWeight` shouldn't change
   191	    if (!T::IsPlaylistME(T::GetPlaylist())) return 1;
   192	    // If this playlist is ME RHC, `GetFluxAndCVWeight` shouldn't change
   193	    // This will change if/when RHC correlated fluxes become available
   194	    //if (!T::isFHC()) return 1; Commented out now that RHC correlated fluxes are available as in Amit's CVS commits 1.19/20 -David L.
   195	
   196	    const double Enu    = T::GetDouble("mc_incomingE")*1e-3;
   197	    const int nu_pdg = T::GetInt("mc_incoming");
   198	    int universe = T::m_nsigma < 0 ? 0 : 1;
   199	    // This universes's flux weight is modified because the flux is strongly correlated with the muon energy scale.
   200	    // The fluxes used for these systematic universes were extracted as part of Amit's wiggle studies.
   201	    double wgtMod = PlotUtils::flux_reweighter(
   202	      T::GetPlaylist(),nu_pdg,T::UseNuEConstraint(),T::GetNFluxUniverses()).GetSysUniFluxWeightCorrection(Enu,nu_pdg,"Muon_Energy_MINOS",universe); // Putting in Amit's CVS commit 1.19/20 changes -David L.
   203	    return wgtMod;
   204	  }
