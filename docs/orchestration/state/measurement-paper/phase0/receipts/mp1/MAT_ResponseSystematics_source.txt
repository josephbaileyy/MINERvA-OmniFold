   344	template <class T>
   345	double ResponseUniverse<T>::GetRecoilShift() const {
   346	  double shift_val = 0;
   347	  if( m_use_new_part_resp )
   348	  {
   349	    //ID only
   350	    //Total recoil
   351	    //  part_response_total_recoil_passive_id/od
   352	    //  part_response_container_total_recoil_passive_id/od
   353	    //Particle recoil
   354	    //  part_response_recoil_passive_<response_name>_id/od
   355	    //  part_response_container_recoil_passive_<response_name>_id/od
   356	    double total_passive_recoil = T::GetDouble("part_response_total_recoil_passive_id");
   357	    double part_passive_recoil  = T::GetDouble(m_particle_passive_name.c_str()); 
   358	     
   359	    for( std::vector<int>::const_iterator idx = T::m_non_cal_idx.begin(); idx != T::m_non_cal_idx.end(); ++idx  )
   360	    {
   361	      total_passive_recoil -= T::GetVecElem("part_response_container_total_recoil_passive_id", *idx);
   362	      part_passive_recoil  -= T::GetVecElem( m_container_particle_passive_name.c_str(), *idx);
   363	    }
   364	      
   365	    shift_val = T::m_nsigma*( (part_passive_recoil/total_passive_recoil)*m_frac_unc )*T::GetCalRecoilEnergy();
   366	  }
   367	  else
   368	  { //std::cout<< "Using err branches" << std::endl;
   369	    //std::cout<< m_branch_name.c_str() << std::endl;
   370	    //double recoil_E_shift = T::GetDouble(m_branch_name.c_str());
   371	    double recoil_E_shift = 0;
   372	    if(m_p4){
